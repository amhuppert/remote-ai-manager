I've added the polling mechanism using useEffect with a cleanup interval. The flow is:

1. Submit prompt → POST to /api/.../prompt → receive runId
2. Start polling /api/runs/:runId at 500ms intervals
3. On "succeeded": display finalText, refresh history
4. On "failed": show error state 

const pollRun = useCallback(async (runId: string) => {
  const interval = setInterval(async () => {
    const res = await fetch(`/api/runs/${runId}`)
    const run = await res.json()
    if (run.status !== 'running') {
      clearInterval(interval)
      if (run.status === 'succeeded') {
        setMessages(prev => [...prev, {
          role: 'assistant', text: run.finalText
        }])
      }
    }
  }, 500)
  return () => clearInterval(interval)
}, [])
